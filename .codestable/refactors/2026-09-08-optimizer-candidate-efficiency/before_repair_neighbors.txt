from __future__ import annotations

import random
from dataclasses import dataclass
from datetime import datetime
from itertools import zip_longest
from typing import Any, Dict, Iterator, List, Tuple

from core.infrastructure.errors import ValidationError


@dataclass(frozen=True)
class RepairNeighborhood:
    order: Tuple[str, ...]
    moves: Tuple[Tuple[str, int, int], ...]

    def neighbors(self) -> Iterator[Tuple[str, List[str]]]:
        for kind, source, target in self.moves:
            order = list(self.order)
            if kind == "adjacent_swap":
                order[source], order[target] = order[target], order[source]
            else:
                order.insert(target, order.pop(source))
            yield kind, order


def build_repair_neighborhood(
    candidate: Dict[str, Any], *, operations: List[Any], metrics_by_op_id: Dict[int, Dict[str, Any]],
    start_dt: datetime, seed: int,
) -> RepairNeighborhood:
    mutable_batches = {str(op.batch_id) for op in operations}
    order = tuple(batch for batch in candidate["decoded_batch_order"] if batch in mutable_batches)
    if len(order) != len(set(order)) or set(order) != mutable_batches:
        raise ValidationError(
            "GraphReady repair 的已解码批次与可变范围不一致。", field="graph_ready_elite_repair",
            details={"reason": "graph_ready_repair_scope_mismatch"},
        )
    signals = _batch_signals(candidate, operations=operations, metrics_by_op_id=metrics_by_op_id, start_dt=start_dt)
    risky = sorted(range(len(order)), key=lambda i: (
        -signals[order[i]]["tardy"], -signals[order[i]]["due_pressure"],
        -signals[order[i]]["critical"], i,
    ))
    risky = [i for i in risky if any(signals[order[i]][key] > 0 for key in ("tardy", "due_pressure", "critical"))][:3]
    adjacent = list(range(max(len(order) - 1, 0)))
    random.Random(int(seed)).shuffle(adjacent)
    adjacent.sort(key=lambda i: not (i in risky or i + 1 in risky))
    swaps = [("adjacent_swap", i, i + 1) for i in adjacent]
    inserts = [("single_insert", i, j) for i in risky
               for j in sorted({0, max(i - 2, 0), min(i + 2, len(order) - 1), len(order) - 1}) if i != j]
    boundary = _boundary_moves(order, signals)
    moves = tuple(move for group in zip_longest(swaps, inserts, boundary) for move in group if move is not None)
    return RepairNeighborhood(order=order, moves=moves)


def repair_priority_context(context: Dict[str, Any], *, operations: List[Any], order: List[str]) -> Dict[str, Any]:
    batches = {str(op.batch_id) for op in operations}
    if len(order) != len(set(order)) or set(order) != batches:
        raise ValidationError(
            "GraphReady repair 优先决策必须覆盖且只覆盖可变批次。", field="graph_ready_elite_repair",
            details={"reason": "graph_ready_repair_scope_mismatch"},
        )
    ranks = {batch: float(i) for i, batch in enumerate(order)}
    out = dict(context)
    # Only the ready priority changes; fixed IDs and precedence maps stay intact.
    out["graph_priority_key_by_op_id"] = {int(op.id): (ranks[str(op.batch_id)],) for op in operations}
    return out


def _batch_signals(
    candidate: Dict[str, Any], *, operations: List[Any], metrics_by_op_id: Dict[int, Dict[str, Any]], start_dt: datetime,
) -> Dict[str, Dict[str, float]]:
    ends: Dict[str, float] = {}
    for result in candidate["results"]:
        batch = str(result.batch_id)
        ends[batch] = max(ends.get(batch, 0.0), (result.end_time - start_dt).total_seconds() / 3600.0)
    signals: Dict[str, Dict[str, float]] = {}
    for op in operations:
        batch = str(op.batch_id)
        metric = metrics_by_op_id[int(op.id)]
        row = signals.setdefault(batch, {key: 0.0 for key in ("tardy", "due_pressure", "critical", "saveability", "sacrifice")})
        row["tardy"] = max(row["tardy"], ends.get(batch, 0.0) - float(metric["due_deadline_hours"]))
        for target, source in (("due_pressure", "due_pressure"), ("saveability", "saveability"), ("sacrifice", "sacrifice_penalty")):
            row[target] = max(row[target], float(metric[source]))
        row["critical"] = max(row["critical"], float(metric["is_on_critical_path"]))
    return signals


def _boundary_moves(order: Tuple[str, ...], signals: Dict[str, Dict[str, float]]) -> List[Tuple[str, int, int]]:
    tardy = [i for i, batch in enumerate(order) if signals[batch]["tardy"] > 0]
    if not tardy:
        return []
    boundary = tardy[0]
    saveable = max(tardy, key=lambda i: (signals[order[i]]["saveability"], -i))
    sacrifice = max(range(len(order)), key=lambda i: (signals[order[i]]["sacrifice"], -i))
    return [("tardy_boundary_move", i, j) for i, j in
            ((saveable, max(boundary - 1, 0)), (sacrifice, len(order) - 1)) if i != j]
